        // Audio Toggle logic
        function toggleAudio() {
            const video = document.getElementById('bgVideo');
            const icon = document.getElementById('audioIcon');
            const text = document.getElementById('audioText');


            if (video.muted) {
                video.muted = false;
                icon.textContent = '🔊';
                text.textContent = 'Mute Sound';
            } else {
                video.muted = true;
                icon.textContent = '🔇';
                text.textContent = 'Unmute Sound';
            }
        }


        // Simple submission handler for demo
        function handleSubscribe(e) {
            e.preventDefault();
            const input = document.getElementById('emailInput');
            const feedback = document.getElementById('formFeedback');
            if (input.value) {
                feedback.style.display = 'block';
                input.value = '';
                setTimeout(() => {
                    feedback.style.display = 'none';
                }, 5000);
            }
        }

Option 1: Cloudinary Direct Upload (Seamless In-App Upload)
With Cloudinary, photos upload instantly right inside your webpage without redirecting the user, and they land straight in a designated Cloudinary media folder.
Setup Steps (5 minutes for Peggy's Granddaughter or Agy):
1. Create a free account at Cloudinary.com.
2. Go to Settings → Upload → Add Upload Preset.
3. Set Mode to Unsigned (this allows family members to upload directly from their phones without needing a password).
4. Set the Folder Name to Darden_Reunion_Archives.
5. Copy your Cloud Name and Upload Preset Name.
JavaScript Code Update:
Replace the handlePhotoSubmit script function near the bottom of your HTML with this direct API call:
async function handlePhotoSubmit(e) {
  e.preventDefault();
  
  const cloudName = "YOUR_CLOUDINARY_CLOUD_NAME"; // Replace with Cloudinary Cloud Name
  const uploadPreset = "YOUR_UPLOAD_PRESET";      // Replace with Unsigned Preset Name
  
  const branchTarget = document.getElementById('modalBranchTarget').value;
  const fileInput = e.target.querySelector('input[type="file"]');
  const caption = e.target.querySelector('textarea').value;
  const feedback = document.getElementById('modalFeedback');


  if (!fileInput.files) return;


  feedback.textContent = "Uploading photo to reunion archives...";
  feedback.style.display = 'block';


  const formData = new FormData();
  formData.append('file', fileInput.files);
  formData.append('upload_preset', uploadPreset);
  formData.append('context', `branch=${branchTarget}|caption=${caption}`);


  try {
    const response = await fetch(`https://api.cloudinary.com/v1_1/${cloudName}/image/upload`, {
      method: 'POST',
      body: formData
    });
    
    if (response.ok) {
      feedback.textContent = "✅ Photo successfully uploaded to the Darden Reunion Archives!";
      setTimeout(() => { closeUploadModal(); }, 3000);
    } else {
      feedback.textContent = "❌ Upload failed. Please try again.";
    }
  } catch (err) {
    feedback.textContent = "❌ Connection error. Please check your internet.";
  }
}
